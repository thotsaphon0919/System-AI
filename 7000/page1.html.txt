<script>
let state=null;
let showId=null;
let selected={type:null,pageId:null,itemId:null};

const $=id=>document.getElementById(id);

function toast(msg){
  const t=$("toast");
  t.textContent=msg;
  t.classList.add("show");
  setTimeout(()=>t.classList.remove("show"),1700);
}

async function api(url,opt={}){
  const res=await fetch(url,{headers:{"Content-Type":"application/json",...(opt.headers||{})},...opt});
  if(!res.ok)throw new Error(await res.text());
  return await res.json();
}

async function load(){
  state=await api("/api/state");
  const m=location.pathname.match(/^\/show\/([^\/]+)/);
  showId=m?decodeURIComponent(m[1]):null;
  document.body.classList.toggle("show-mode",!!showId);
  render();
}

function page(){
  return state.pages.find(p=>p.id===showId)||null;
}

function mediaHtml(obj,cls="media"){
  if(obj.media_url&&obj.media_type==="video")return `<div class="${cls}"><video src="${obj.media_url}" muted autoplay loop playsinline></video></div>`;
  if(obj.media_url)return `<div class="${cls}"><img src="${obj.media_url}"></div>`;
  return '<div class="plus">+</div>';
}

function render(){
  if(showId) return renderShow();
  $("modeText").textContent="ทุกช่อง = หน้าโชว์ของย่อย";
  $("head").innerHTML="<h2>หน้าโชว์ทั้งหมด</h2><p>แตะช่องเพื่อดูของย่อย · กด ⚙ เพื่อตั้งค่า</p>";
  $("addPageBtn").textContent="+ หน้าโชว์";
  $("uploadPagesBtn").textContent="+ รูปเป็นหน้า";
  $("backBtn").textContent="จัดหน้า";
  const grid=$("grid");
  grid.className="grid";
  grid.innerHTML="";
  state.pages.forEach(pg=>grid.appendChild(pageCard(pg)));
}

function renderShow(){
  const pg=page();
  if(!pg){ location.href="/"; return; }
  $("modeText").textContent="หน้าโชว์ของย่อย";
  $("addPageBtn").textContent="← หน้าแรก";
  $("uploadPagesBtn").textContent="อัปโหลดรูปเป็นหน้า";
  $("backBtn").textContent="เปิดลิงก์";
  $("head").innerHTML="";
  const grid=$("grid");
  grid.className="";
  grid.innerHTML=`
    <div class="hero">
      ${mediaHtml(pg)}
      <button class="gear" type="button" id="pageGear">⚙</button>
      <div class="heroTitle"><h2>${esc(pg.title)}</h2><p>${esc(pg.subtitle||"")}</p></div>
    </div>
    <div class="itemList" id="itemList"></div>
  `;
  $("pageGear").addEventListener("click",e=>{e.stopPropagation();openPageGear(pg.id)});
  document.querySelector(".hero").addEventListener("click",e=>{
    if(e.target.closest(".gear"))return;
    if(pg.media_url)openViewer(pg);
  });
  const list=$("itemList");
  pg.items.forEach((it,idx)=>list.appendChild(itemCard(pg,it,idx)));
}

function pageCard(pg){
  const el=document.createElement("div");
  el.className="card";
  el.innerHTML=`
    ${mediaHtml(pg)}
    <button class="gear" type="button">⚙</button>
    <div class="body"><h3>${esc(pg.title)}</h3><p>${esc(pg.subtitle||"หน้าโชว์ของย่อย")}</p></div>
  `;
  el.addEventListener("click",e=>{
    if(e.target.closest(".gear"))return;
    location.href="/room/"+encodeURIComponent(pg.id);
  });
  el.querySelector(".gear").addEventListener("click",e=>{
    e.preventDefault();e.stopPropagation();openPageGear(pg.id);
  });
  return el;
}

function itemCard(pg,it,idx){
  const el=document.createElement("div");
  el.className="item";
  let mini="";
  if(it.media_url&&it.media_type==="video")mini=`<div class="mini"><video src="${it.media_url}" muted autoplay loop playsinline></video></div>`;
  else if(it.media_url)mini=`<div class="mini"><img src="${it.media_url}"></div>`;
  el.innerHTML=`
    ${mini}
    <button class="gear" type="button">⚙</button>
    <div class="itemBody"><b>no.${idx+1}</b><span>${esc(it.title||"")}</span></div>
  `;
  el.addEventListener("click",e=>{
    if(e.target.closest(".gear"))return;
    useItem(pg.id,it.id);
  });
  el.querySelector(".gear").addEventListener("click",e=>{
    e.preventDefault();e.stopPropagation();openItemGear(pg.id,it.id);
  });
  return el;
}

function esc(x){
  return String(x||"").replace(/[&<>"]/g,m=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[m]));
}

function getPage(id){return state.pages.find(p=>p.id===id)}
function getItem(pid,iid){return getPage(pid).items.find(x=>x.id===iid)}

function openPageGear(pid){
  selected={type:"page",pageId:pid,itemId:null};
  const pg=getPage(pid);
  fillForm(pg,"ตั้งค่าหน้าโชว์");
}

function openItemGear(pid,iid){
  selected={type:"item",pageId:pid,itemId:iid};
  const it=getItem(pid,iid);
  fillForm(it,"ตั้งค่าของย่อย");
}

function fillForm(obj,title){
  $("drawerTitle").textContent=title;
  $("fTitle").value=obj.title||"";
  $("fSubtitle").value=obj.subtitle||"";
  $("fDetail").value=obj.detail||"";
  $("fLink").value=obj.link_url||"";
  $("fPublish").value=obj.publish||"private";
  $("fEnabled").checked=obj.enabled!==false;
  $("fFile").value="";
  showPreview(obj);
  $("back").classList.add("show");
  $("drawer").classList.add("show");
}

function showPreview(obj){
  const p=$("preview");
  if(obj.media_url&&obj.media_type==="video")p.innerHTML=`<video src="${obj.media_url}" controls playsinline></video>`;
  else if(obj.media_url)p.innerHTML=`<img src="${obj.media_url}">`;
  else p.textContent="ยังไม่มีสื่อ";
}

function closeDrawer(){
  $("back").classList.remove("show");
  $("drawer").classList.remove("show");
}

async function saveTarget(){
  const body={
    title:$("fTitle").value,
    subtitle:$("fSubtitle").value,
    detail:$("fDetail").value,
    link_url:$("fLink").value,
    publish:$("fPublish").value,
    enabled:$("fEnabled").checked
  };
  if(selected.type==="item"){
    state=await api(`/api/pages/${selected.pageId}/items/${selected.itemId}`,{method:"PATCH",body:JSON.stringify(body)});
  }else{
    state=await api(`/api/pages/${selected.pageId}`,{method:"PATCH",body:JSON.stringify(body)});
  }
  toast("บันทึกแล้ว");
  render();
