function mediaHtml(url,type,emptyClass){
  if(!url) return `<div class="${emptyClass}">+</div>`;
  if(type==="video" || url.toLowerCase().match(/\.(mp4|webm|mov)$/)){

  return `<img src="${url}">`;
}

function pickFile(uploadUrl){
  const input=document.createElement("input");
  input.type="file";
  input.accept="image/*,video/*";
  input.onchange=async()=>{
    const f=input.files[0];
    if(!f) return;
